~~~bash
#!/bin/bash

LOCKFILE="/tmp/rsync.lock"
LOGFILE="/var/log/rsync_backup.log"
MIN_SPACE_MB=10240  # 最小剩余空间检查，单位MB
RETRIES=3  # 重试次数
DELAY=10   # 重试延迟
API_URL="http://1.94.145.39:3000/send_private_msg"
USER_ID=2996014561  # 替换为目标用户 ID

# 获取当前日期
DATE=$(date +"%Y-%m-%d")

# 本地备份的日期目录路径
BACKUP_ROOT="/mnt/sda2/server_data_backup/$DATE"
DOCKER_BACKUP_DIR="$BACKUP_ROOT/docker_data"
MD_BACKUP_DIR="$BACKUP_ROOT/md"

echo "=== Backup started at $(date) ===" | tee -a $LOGFILE

check_space() {
    AVAIL_SPACE=$(df /mnt/sda2 | awk 'NR==2 {print $4}')  # 获取可用空间(KB)
    if [ $AVAIL_SPACE -lt $((MIN_SPACE_MB*1024)) ]; then
        echo "Not enough space for backup. Available: $((AVAIL_SPACE/1024)) MB" | tee -a $LOGFILE
        exit 1
    fi
}

send_result() {
    local result=$1
    curl -X POST $API_URL \
    -H "Content-Type: application/json" \
    -d '{
        "user_id": '$USER_ID',
        "message": {
            "type": "text",
            "data": {
                "text": "'"$result"'"
            }
        }
    }'
}

backup_task() {
    local src=$1
    local dest=$2
    local task_name=$3

    echo "=== Start $task_name at $(date) ===" | tee -a $LOGFILE
    mkdir -p $dest  # 创建当天备份目录
    for ((i=1; i<=RETRIES; i++)); do
        RESULT=$(flock -n -E 1 $LOCKFILE -c "rsync -avzP --timeout=60 --delete $src $dest 2>&1")
        echo "$RESULT" | tee -a $LOGFILE
        if [ $? -eq 0 ]; then
            echo "=== Finished $task_name at $(date) ===" | tee -a $LOGFILE
            send_result "Backup task: $task_name completed successfully."
            break
        elif [ $i -lt $RETRIES ]; then
            echo "Retrying $task_name in $DELAY seconds..." | tee -a $LOGFILE
            sleep $DELAY
        else
            echo "$task_name failed after $RETRIES attempts." | tee -a $LOGFILE
            send_result "Backup task: $task_name failed after $RETRIES attempts."
        fi
    done
}

# 删除超过5天的备份
delete_old_backups() {
    echo "=== Deleting backups older than 5 days ===" | tee -a $LOGFILE
    find /mnt/sda2/server_data_backup/ -maxdepth 1 -type d -mtime +5 -exec rm -rf {} \; | tee -a $LOGFILE
}

# 检查磁盘空间
check_space

# 删除旧备份
delete_old_backups

# 备份任务
backup_task "root@1.94.145.39:/vw-data/" "/opt/docker_data/vaultwarden/" "vaultwarden data backup"
backup_task "root@190.92.233.153:/opt/docker_data/alist/" "/opt/md/" "alist MD data backup"
backup_task "root@124.223.67.235:/opt/docker_data/elasticsearch/" "/opt/docker_data/elasticsearch/" "elasticsearch data backup"
backup_task "root@124.223.67.235:/opt/docker_data/mysql/" "/opt/docker_data/mysql/" "mysql data backup"
backup_task "root@124.223.67.235:/opt/docker_data/nacos/" "/opt/docker_data/nacos/" "nacos data backup"
backup_task "root@124.223.67.235:/opt/docker_data/redis/" "/opt/docker_data/redis/" "redis data backup"
backup_task "/opt/docker_data/" "$DOCKER_BACKUP_DIR" "local docker data backup"
backup_task "/opt/md/" "$MD_BACKUP_DIR" "local MD backup"

echo "=== Backup completed at $(date) ===" | tee -a $LOGFILE
send_result "All backup tasks completed."
~~~
